    
    
WITH base_data AS (
  SELECT
    CASE 
      WHEN MAIN_BILLER_ID IN (14, 175) THEN 14
      WHEN MAIN_BILLER_ID IN (13, 160) THEN 13
      ELSE MAIN_BILLER_ID
    END AS GROUPED_BILLER_ID,
    TOTAL_AMOUNT,
    CREATED_AT,
    -- Set Thursday as start of the week
    TRUNC(CREATED_AT, 'IW') + 1 AS WEEK_START
  FROM REP.TRANSACTIONS_NBP_TEMP tnt 
  WHERE MAIN_BILLER_ID IN (13, 14, 15, 16, 175, 160)
    AND SERVICE_ID NOT IN (235, 236,1783, 1784, 1434, 1435)
),
weekly_agg AS (
  SELECT
    WEEK_START,
    GROUPED_BILLER_ID AS MAIN_BILLER_ID,
    SUM(TOTAL_AMOUNT) AS TOTAL_AMOUNT,
    COUNT(*) AS TRANSACTION_COUNT,
    EXTRACT(YEAR FROM WEEK_START) AS YEAR,
    EXTRACT(MONTH FROM WEEK_START) AS MONTH,
    TO_CHAR(WEEK_START, 'IW') AS WEEK_NUMBER
  FROM base_data
  GROUP BY WEEK_START, GROUPED_BILLER_ID
)
SELECT
  WEEK_START,
  MAIN_BILLER_ID,
  YEAR,
  MONTH,
  WEEK_NUMBER,
  TOTAL_AMOUNT,
  TRANSACTION_COUNT,

  -- Last week
  LAG(TOTAL_AMOUNT, 1) OVER (PARTITION BY MAIN_BILLER_ID ORDER BY WEEK_START) AS LAST_WEEK_TOTAL_AMOUNT,
  LAG(TRANSACTION_COUNT, 1) OVER (PARTITION BY MAIN_BILLER_ID ORDER BY WEEK_START) AS LAST_WEEK_TRANSACTION_COUNT,

  -- Last 2 weeks ago
  LAG(TOTAL_AMOUNT, 2) OVER (PARTITION BY MAIN_BILLER_ID ORDER BY WEEK_START) AS LAST_2WEEK_TOTAL_AMOUNT,
  LAG(TRANSACTION_COUNT, 2) OVER (PARTITION BY MAIN_BILLER_ID ORDER BY WEEK_START) AS LAST_2WEEK_TRANSACTION_COUNT,

  -- Next week
  LEAD(TOTAL_AMOUNT, 1) OVER (PARTITION BY MAIN_BILLER_ID ORDER BY WEEK_START) AS NEXT_WEEK_TOTAL_AMOUNT

FROM weekly_agg
ORDER BY MAIN_BILLER_ID, WEEK_START;
